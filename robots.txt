User-agent: *
Allow: /

# Query-parameterised template URLs are prototype-only. In production these
# become real paths (/services/..., /usa/california/...) and these rules can be
# removed. Until then, block them so the same page is not indexed many times
# under different query strings.
Disallow: /*?service=
Disallow: /*?industry=
Disallow: /*?city=
Disallow: /*?state=

# Filter and search parameters produce no unique indexable content.
Disallow: /*?category=
Disallow: /*?technology=
Disallow: /*?type=
Disallow: /*?q=

# Never index the error page.
Disallow: /404.html

# Assets must stay crawlable — blocking CSS or JS prevents search engines
# rendering the page as a visitor sees it.
Allow: /assets/
